# 1. 공구집 일일 퀘스트 — 전체 19단계
# ════════════════════════════════════════════════════════════════════════════
#
#  #   단계 id              하는 일                             비고
#  ──  ───────────────────  ──────────────────────────────────  ──────────────────────────
#  ▼ 1-A. 자동 구간 — `python3 -m gonggu.daily` 한 줄이 1~14를 순서대로
#   1  update_gonggu_stage  공구 상태(시작전/진행중/종료) 갱신   README §6. 먼저 와야 9·10이 정확
#   2  fetch_source         원본 수집                            README §1. DAYS_BACK=7
#   3  fetch_yt_ppl         유튜브 PPL 원본 수집                 README §8-1. 나머지와 완전 독립
#   4  classify             LLM#1 공구 분류                      README §2. CONCURRENCY=200
#   5  classify_yt_ppl      유튜브 PPL 공구 판별                 README §8-2
#   6  transform            보수적 게이트링                      README §3
#   7  resolve_links        링크 해석 (Playwright — uc 아님)     README §4. 하루 중 제일 김
#   8  load                 DB 적재                              README §5. 7 다음이어야 함
#   9  rescan_inprogress    진행중인데 못 찾은 링크 재탐색       README §7
#  10  backfill_period      공구기간 백필 (Tier0 인포크→Tier1 몰) README §9
#  11  sync_emails          크리에이터 이메일 → hifen DB         실패해도 계속
#  12  uc_gate              uc 신뢰 관문 (필요할 때만 워밍업)     실패해도 계속 / gonggu/pipeline/uc_gate.py
#  13  reverify_uc          차단 unresolved를 uc로 재검증        실패해도 계속 / 시간예산 30분
#  14  maintenance          하우스키핑(컴팩션/로테이션)          실패해도 계속 / 항상 마지막
#
#  ▼ 1-B. 수동 구간 — 상세수집. daily가 끝난 뒤(링크가 done인 상품이 있어야 대상이 생김)
#  15  enrich_detail        fast 자사몰 대량 보강                무인·병렬·안정
#  16  warmup_naver_uc      uc 신뢰 워밍업                       ⚠ 신뢰가 깨졌을 때만
#  17  crawl_stage          uc 크롤링만                          사람이 곁에서
#  18  llm_stage            LLM만 (브라우저 무관)                고동시성
#  19  load_stage           DB 반영만
#
#  ⚠ 15~19가 아직 daily에 없는 이유: 17(crawl_stage)이 uc 창을 대량으로 쓰는 구간이라 통합
#    리스크가 제일 크다. 12(uc_gate)가 실전에서 한 사이클 검증된 뒤에 얹기로 했다(2026-08-20).


# ────────────────────────────────────────────────────────────────────────────
# 1-A. 자동 구간 (1~14) — daily 한 줄
# ────────────────────────────────────────────────────────────────────────────

python3 -m gonggu.daily                                 # ★ 이거 하나면 1~14 끝
python3 -m gonggu.daily --list                          # 순서/기본값/속성 확인

# ── 알아둘 것 세 개 ──────────────────────────────────────────────────────────
#  (1) `rm -rf ~/.gonggu_uc_profile`이 사라졌다. 예전엔 매일 uc 신뢰를 버리고 다시 쌓았는데
#      대부분의 날은 프로필이 멀쩡해서 낭비였다. 12(uc_gate)가 먼저 비대화형으로 "지금 뚫리는가"를
#      확인하고(창이 잠깐 떴다 닫힘, 프로필이 아예 없으면 창도 안 띄움) 살아있으면 그대로 통과,
#      죽었을 때만 초기화 + 워밍업을 띄운다.
#  (2) 워밍업을 띄울지는 stdin이 TTY인지로 정한다. cron/nohup으로 돌리면 워밍업 없이 13만
#      건너뛰고 나머지는 다 돈다. TTY인데 자리를 비우면 600초 뒤 같은 처리(UC_WARMUP_TIMEOUT_SEC).
#  (3) 실패 정책이 갈린다. 1~10 본줄기는 뒤 단계가 앞 결과에 의존하므로 즉시 중단(+`--from` 안내).
#      11~14는 서로 독립이라 실패해도 계속 가고 마지막 요약에 ✗로 모아 보여준다(+`--only` 안내).

# ── 구간 지정: --from <단계> / --until <단계, 포함> / --only <단계> ───────────
#   이름은 예전 그대로다(테스트로 못박아둠 — tests/test_daily_stages.py).
#   적용 순서는 --from → --until → --only.

#   ▸ 하루를 두 번에 나눠 돌리는 패턴 (--until의 존재 이유) — 긴 무인 단계를 오프피크에 먼저
#     돌려두고, 사람이 붙어야 하는 uc 구간은 자리에 있을 때 따로:
python3 -m gonggu.daily --from rescan_inprogress --until backfill_period   # 1차: 무인, 오프피크
UC_WARMUP_TIMEOUT_SEC=1800 python3 -m gonggu.daily --from sync_emails      # 2차: 12가 바로 뜸

#   ▸ 중단된 지점부터 이어서
python3 -m gonggu.daily --from update_gonggu_stage      #  1. 공구 상태 갱신부터(= 전체와 동일)
python3 -m gonggu.daily --from fetch_source             #  2. 원본 수집부터
python3 -m gonggu.daily --from fetch_yt_ppl             #  3. 유튜브 PPL 수집부터
python3 -m gonggu.daily --from classify                 #  4. LLM 공구 분류부터
python3 -m gonggu.daily --from classify_yt_ppl          #  5. 유튜브 PPL 판별부터
python3 -m gonggu.daily --from transform                #  6. 게이트링부터
python3 -m gonggu.daily --from resolve_links            #  7. 링크 해석부터
python3 -m gonggu.daily --from load                     #  8. DB 적재부터
python3 -m gonggu.daily --from rescan_inprogress        #  9. 진행중 재탐색부터
python3 -m gonggu.daily --from backfill_period          # 10. 공구기간 백필부터
python3 -m gonggu.daily --from sync_emails              # 11. 뒷단 보강만(11~14)
python3 -m gonggu.daily --from uc_gate                  # 12. uc 관문부터(12~14)
python3 -m gonggu.daily --from maintenance              # 14. 하우스키핑만 남았을 때

#   ▸ 한 단계만 / 어디까지만
python3 -m gonggu.daily --only resolve_links            # 링크 해석만 다시
python3 -m gonggu.daily --only load                     # 적재만 다시
python3 -m gonggu.daily --only reverify_uc              # uc 재검증만(게이트 없이 — 워밍업이 돼 있다는 전제)
python3 -m gonggu.daily --until load                    # 적재까지만 하고 멈춤
python3 -m gonggu.daily --from load --until load        # (= --only load)

# ── ⚠ 13(reverify_uc)에 대해 알아야 할 것 — "uc 큐가 왜 안 줄지?" ────────────
#   이건 큐를 비우는 단계가 아니라 매일 30분씩 갉는 단계다. 2026-08-20 실측:
#       유입: resolve 한 번이 만드는 uc 후보 하루 약 436건(로그인월_차단 69 + 재검증 중 차단 367)
#       배수: uc는 uc_engine._lock으로 전역 직렬 — 동시성을 올려도 uc 페치는 한 번에 하나다.
#             30분 예산이면 건당 15초→120건 / 30초→60건 / 75초(타임아웃)→24건
#   즉 유입 436 vs 배수 24~120이라 큐는 계속 자란다(8/19 702건 → 8/20 2,301건).
#   손익분기는 건당 30초 기준 하루 3.3시간 — 데일리 한 단계에 넣을 시간이 아니다.
#   UC_TIME_BUDGET_SEC=1800의 역할은 큐 감축이 아니라 안전밸브다(uc가 죽어도 데일리가 묶이는
#   최대치를 30분으로 못박음). 진짜 해법은 병렬화 — 1-B의 UC_PROFILE 샤딩과 같은 패턴을
#   여기 적용하면 프로필 N개로 N배. 건당 실제 소요가 나온 뒤에 검토한다.
#   실행하면 세 숫자를 같이 찍어준다: 대상 N건 / 예산 내 M건 시도 / 큐 잔량 + 유입 추정.

# (17) uc 크롤링만 — 3단계 분리 1단계. 크롤링 결과(detail_crawled*.jsonl)는 LLM 결과와 별도
#      파일로 영구 보존 — 나중에 프롬프트 고쳐 LLM만 다시 돌려도(FORCE_RELLM=1) 재크롤링 불필요.

# (18) LLM만 — 브라우저와 무관, 고동시성(기본 60)으로 크롤링 결과를 몰아서 처리
LLM_STAGE_CONCURRENCY=60 python3 -m gonggu.enrich_detail.llm_stage
FORCE_RELLM=1 python3 -m gonggu.enrich_detail.llm_stage  # 프롬프트 수정 후 이미 처리된 것까지 전부 재처리

# (19) DB 반영만
python3 -m gonggu.enrich_detail.load_stage

# ── (17) 샤딩 버전 — 백로그가 커서 위 (17)이 너무 느릴 때만. 쓸 수도 안 쓸 수도 있음 ──
#   신뢰 쿠키 프로필을 N벌 만들어 진짜 병렬(각자 다른 UC_PROFILE). crawl_stage.py가 2026-08-18
#   부터 샤드 파일 전체(detail_crawled*.jsonl)를 합쳐 중복 체크하므로, 샤딩했다 안 했다 오가도
#   이미 크롤링된 건은 안전하게 건너뛴다.

UC_PROFILE=$HOME/.gonggu_uc_profile_2 python3 -m gonggu.enrich_detail.warmup_naver_uc

UC_LOGIN_WAIT=0 DETAIL_MODE=uc UC_PROFILE=$HOME/.gonggu_uc_profile_1 SHARD_COUNT=2 SHARD_INDEX=0 python3 -m gonggu.enrich_detail.crawl_stage
UC_LOGIN_WAIT=0 DETAIL_MODE=uc UC_PROFILE=$HOME/.gonggu_uc_profile_2 SHARD_COUNT=2 SHARD_INDEX=1 python3 -m gonggu.enrich_detail.crawl_stage
